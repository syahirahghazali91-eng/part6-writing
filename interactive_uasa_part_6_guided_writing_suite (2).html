Google Apps Script Setup Guide

2-Way Cloud Sync for Google Sheets

  1. Create a new Google Sheet.
  2. Go to Extensions > Apps Script.
  3. Paste the script snippet below and click Save.
  4. Click Deploy > New deployment. Choose type Web app.
  5. Set "Execute as" to Me and "Who has access" to Anyone (CRITICAL).
  6. Deploy and paste the Web App URL into the Teacher Dashboard box!
function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: [] })).setMimeType(ContentService.MimeType.JSON);
    var headers = data[0];
    var rows = [];
    for (var i = 1; i < data.length; i++) {
      var row = {};
      for (var j = 0; j < headers.length; j++) { row[headers[j]] = data[i][j]; }
      rows.push(row);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success', data: rows })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Timestamp", "Name", "Class", "Avatar", "Topic", "Score", "WordCount", "EmailDraft"]);
    }
    var data = {};
    if (e && e.postData && e.postData.contents) {
      try { data = JSON.parse(e.postData.contents); } catch(err) { data = e.parameter || {}; }
    } else if (e && e.parameter) { data = e.parameter; }
    
    sheet.appendRow([
      new Date(),
      data.name || '',
      data.className || '',
      data.avatar || '',
      data.topic || '',
      data.score !== undefined ? data.score : 0,
      data.wordCount || 0,
      data.emailDraft || ''
    ]);
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
